string MACRO_DOC_PATH = "C:\\Path\\To\\YourMacroDoc.docm" // Path to your macro file
OleAutoObj objMacroDoc = null

bool startWordInstance() {
    // 1. Clean up existing objects
    if (!null objMacroDoc) { oleCloseAutoObject(objMacroDoc); objMacroDoc = null }
    if (!null hDocs)       { oleCloseAutoObject(hDocs);       hDocs = null }
    if (!null hWord) { 
        noError()
        oleMethod(hWord, "Quit")
        lastError()
        oleCloseAutoObject(hWord)
        hWord = null 
    }
    
    // 2. Launch new Word process
    hWord = oleCreateAutoObject("Word.Application")
    if (null hWord) return false
    
    // 3. Performance & Stability Flags
    olePut(hWord, "Visible", false)
    olePut(hWord, "DisplayAlerts", 0)
    olePut(hWord, "ScreenUpdating", false)
    
    // Speed optimization: Turn off proofing & AutoRecover
    OleAutoObj hOptions = null
    oleGet(hWord, "Options", hOptions)
    if (!null hOptions) {
        olePut(hOptions, "CheckSpellingAsYouType", false)
        olePut(hOptions, "CheckGrammarAsYouType", false)
        olePut(hOptions, "SaveInterval", 0)
        olePut(hOptions, "SaveNormalPrompt", false)
        oleCloseAutoObject(hOptions)
    }
    
    oleGet(hWord, "Documents", hDocs)
    
    // 4. RE-OPEN THE MACRO DOCUMENT IN THE NEW WORD PROCESS
    OleAutoArgs macroOpenArgs = create
    put(macroOpenArgs, MACRO_DOC_PATH)
    put(macroOpenArgs, false) // ConfirmConversions = False
    put(macroOpenArgs, true)  // ReadOnly = True
    put(macroOpenArgs, false) // AddToRecentFiles = False
    
    noError()
    oleMethod(hDocs, "Open", macroOpenArgs, objMacroDoc)
    string macroOpenErr = lastError()
    clear(macroOpenArgs)
    
    if (length(macroOpenErr) > 0 || null objMacroDoc) {
        print "[ERROR] Could not load macro document: " macroOpenErr "\n"
        return false
    }
    
    return true
}
