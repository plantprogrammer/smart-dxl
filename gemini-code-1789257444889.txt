// =========================================================================
// HIGH-VOLUME HTML TO PDF CONVERTER (HDOC-FREE & RECYCLING ENABLED)
// =========================================================================

OleAutoObj hWord = null
OleAutoObj hDocs = null

// --- GLOBAL CONFIGURATION ---
const int BATCH_RECYCLE_LIMIT = 15 // Restart Word every 15 files to clear memory leaks
int docCounter = 0

// Target file arrays (Populate with your file paths)
string htmlFiles[] = { "C:\\Docs\\file1.html", "C:\\Docs\\file2.html", "C:\\Docs\\file3.html" }
string pdfFiles[]  = { "C:\\PDFs\\file1.pdf",  "C:\\PDFs\\file2.pdf",  "C:\\PDFs\\file3.pdf"  }
int totalFiles = sizeof(htmlFiles)

// --- HELPER FUNCTION: LAUNCH / RESTART WORD ---
bool startWordInstance() {
    // Safely cleanup old handles if they exist
    if (!null hDocs) { oleCloseAutoObject(hDocs); hDocs = null }
    if (!null hWord) { 
        noError()
        oleMethod(hWord, "Quit")
        lastError()
        oleCloseAutoObject(hWord)
        hWord = null 
    }
    
    // Create new Word instance
    hWord = oleCreateAutoObject("Word.Application")
    if (null hWord) {
        return false
    }
    
    // Performance & Stability Flags
    olePut(hWord, "Visible", false)
    olePut(hWord, "DisplayAlerts", 0)        // 0 = wdAlertsNone
    olePut(hWord, "ScreenUpdating", false)   // Stops screen repainting
    
    oleGet(hWord, "Documents", hDocs)
    return true
}

// Start Word initially
if (!startWordInstance()) {
    ack "Fatal Error: Could not initialize Microsoft Word."
    halt
}

print "Starting batch process of " totalFiles " files...\n"

// --- MAIN PROCESSING LOOP ---
int i
for i in 0 .. (totalFiles - 1) do {
    string srcHtml = htmlFiles[i]
    string destPdf = pdfFiles[i]
    
    // 1. RECYCLE WORD PROCESS EVERY N FILES TO PREVENT MEMORY LEAKS
    if (docCounter >= BATCH_RECYCLE_LIMIT) {
        print "Flushing Word memory heap... Recycling Word instance at index " i "\n"
        if (!startWordInstance()) {
            ack "Fatal Error: Failed to recycle Word instance."
            halt
        }
        docCounter = 0
    }
    
    OleAutoArgs args = null
    
    // 2. OPEN HTML FILE WITHOUT RETURNING AN hDoc OLE OBJECT
    args = create
    put(args, srcHtml) // 1. FileName
    put(args, false)   // 2. ConfirmConversions = False
    put(args, true)    // 3. ReadOnly = True
    put(args, false)   // 4. AddToRecentFiles = False
    
    noError()
    // Notice: No 4th argument receiver (hDoc) passed to oleMethod
    oleMethod(hDocs, "Open", args)
    string openErr = lastError()
    clear(args)
    
    // Brief pause to allow Word's asynchronous HTML renderer to settle
    sleep_(150)
    
    if (length(openErr) > 0) {
        print "[ERROR] Failed to open file: " srcHtml " (" openErr ")\n"
    } else {
        // 3. RUN MACRO DIRECTLY VIA hWord
        // (Macro targets ActiveDocument to format, export PDF, and close)
        args = create
        put(args, "YourMacroName") // Replace with actual Macro name
        
        noError()
        oleMethod(hWord, "Run", args)
        string macroErr = lastError()
        clear(args)
        
        // 4. DETECT RPC / CRASH FAILURES
        if (length(macroErr) > 0) {
            print "[ERROR] Macro failure on index " i ": " macroErr "\n"
            
            // If Word crashed or RPC connection broke, recover automatically
            if (contains(macroErr, "remote procedure call", 0) >= 0 || contains(macroErr, "argument names", 0) >= 0) {
                print "   -> Word RPC crashed! Re-initializing Word server...\n"
                startWordInstance()
                docCounter = 0
                continue // Move to next file
            }
        }
        
        docCounter++
    }
}

// --- CLEANUP AT END OF BATCH ---
if (!null hWord) {
    olePut(hWord, "ScreenUpdating", true)
    oleMethod(hWord, "Quit")
}

if (!null hDocs) oleCloseAutoObject(hDocs)
if (!null hWord) oleCloseAutoObject(hWord)

print "Batch processing completed successfully.\n"
